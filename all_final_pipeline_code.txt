def make_model(
    seed=RANDOM_STATE
):

    return XGBRegressor(

        n_estimators=500,

        learning_rate=0.03,

        max_depth=6,

        min_child_weight=5,

        subsample=0.8,

        colsample_bytree=0.9,

        reg_lambda=1.0,

        objective="reg:squarederror",

        eval_metric="rmse",

        random_state=seed,

        n_jobs=-1,

        tree_method="hist"
    )


# ============================================================
# LEAKAGE-FREE OOF RESIDUALS
# ============================================================
#
# The variance model must NOT be trained on residuals obtained
# from predictions made by a model trained on the same samples.
#
# Therefore, within each OUTER TRAINING SET:
#
#   inner spatial folds
#       ↓
#   OOF mean prediction
#       ↓
#   squared residual
#
# These OOF squared residuals become the target for the
# uncertainty model.
# ============================================================

def build_oof_residual_targets(
    X_train,
    y_train,
    groups_train
):

    inner_cv = GroupKFold(
        n_splits=N_INTERNAL_FOLDS
    )


    oof_prediction = np.full(
        len(X_train),
        np.nan,
        dtype=float
    )


    for inner_fold, (
        inner_train_idx,
        inner_valid_idx
    ) in enumerate(

        inner_cv.split(
            X_train,
            y_train,
            groups=groups_train
        ),

        start=1

    ):

        model = make_model(
            seed=RANDOM_STATE + inner_fold
        )


        X_inner_train = (
            X_train.iloc[
                inner_train_idx
            ]
        )

        y_inner_train = (
            y_train.iloc[
                inner_train_idx
            ]
        )

        X_inner_valid = (
            X_train.iloc[
                inner_valid_idx
            ]
        )


        model.fit(
            X_inner_train,
            y_inner_train
        )


        oof_prediction[
            inner_valid_idx
        ] = model.predict(
            X_inner_valid
        )


    valid_mask = np.isfinite(
        oof_prediction
    )


    if not valid_mask.all():

        raise RuntimeError(
            "OOF residual generation failed."
        )


    residual = (
        y_train.to_numpy()
        -
        oof_prediction
    )


    squared_residual = (
        residual ** 2
    )


    return (
        oof_prediction,
        residual,
        squared_residual
    )


# ============================================================
# FIT FINAL MEAN + UNCERTAINTY MODELS
# ============================================================

def fit_uncertainty_models(
    X_train,
    y_train,
    groups_train
):

    # --------------------------------------------------------
    # OOF residual target
    # --------------------------------------------------------

    (
        oof_prediction,
        oof_residual,
        squared_residual
    ) = build_oof_residual_targets(
        X_train,
        y_train,
        groups_train
    )


    # --------------------------------------------------------
    # Mean model on ALL outer-training observations
    # --------------------------------------------------------

    mean_model = make_model(
        seed=RANDOM_STATE
    )

    mean_model.fit(
        X_train,
        y_train
    )


    # --------------------------------------------------------
    # Clip squared residuals to prevent a few massive errors
    # from completely dominating variance fitting.
    #
    # This clipping threshold is calculated using training data
    # only.
    # --------------------------------------------------------

    upper = np.quantile(
        squared_residual,
        0.99
    )


    variance_target = np.minimum(
        squared_residual,
        upper
    )


    # Log variance target improves numerical stability.
    log_variance_target = np.log1p(
        variance_target
    )


    # --------------------------------------------------------
    # Variance model
    # --------------------------------------------------------

    variance_model = make_model(
        seed=RANDOM_STATE + 100
    )

    variance_model.fit(
        X_train,
        log_variance_target
    )


    return (
        mean_model,
        variance_model,
        upper
    )


# ============================================================
# OUTER SPATIAL CV
# ============================================================

outer_cv = GroupKFold(
    n_splits=N_OUTER_FOLDS
)


fold_results = []

observation_results = []


for fold, (
    train_idx,
    test_idx
) in enumerate(

    outer_cv.split(
        X,
        y,
        groups=groups
    ),

    start=1
):

    print()
    print("============================================================")
    print(
        "OUTER SPATIAL FOLD",
        fold
    )
    print("============================================================")


    X_train = X.iloc[
        train_idx
    ].copy()

    y_train = y.iloc[
        train_idx
    ].copy()

    groups_train = groups.iloc[
        train_idx
    ].copy()


    X_test = X.iloc[
        test_idx
    ].copy()

    y_test = y.iloc[
        test_idx
    ].copy()


    # ========================================================
    # FIT MEAN + VARIANCE MODELS
    # ========================================================

    (
        mean_model,
        variance_model,
        variance_clip
    ) = fit_uncertainty_models(

        X_train,
        y_train,
        groups_train
    )


    # ========================================================
    # TEST PREDICTIONS
    # ========================================================

    mean_prediction = mean_model.predict(
        X_test
    )


    predicted_log_variance = (
        variance_model.predict(
            X_test
        )
    )


    predicted_variance = (
        np.expm1(
            predicted_log_variance
        )
    )


    predicted_variance = np.maximum(
        predicted_variance,
        1e-8
    )


    predicted_sigma = np.sqrt(
        predicted_variance
    )


    # ========================================================
    # CLIP EXTREME UNCERTAINTY FOR STABILITY
    # ========================================================

    max_sigma = np.sqrt(
        max(
            variance_clip,
            1e-8
        )
    )


    predicted_sigma = np.minimum(
        predicted_sigma,
        5.0 * max_sigma
    )


    # ========================================================
    # METRICS
    # ========================================================

    prediction_error = (
        y_test.to_numpy()
        -
        mean_prediction
    )


    absolute_error = np.abs(
        prediction_error
    )


    squared_error = (
        prediction_error ** 2
    )


    r2 = r2_score(
        y_test,
        mean_prediction
    )


    rmse = np.sqrt(
        mean_squared_error(
            y_test,
            mean_prediction
        )
    )


    mae = mean_absolute_error(
        y_test,
        mean_prediction
    )


    # ========================================================
    # COVERAGE
    # ========================================================
    #
    # Approximate 95% prediction interval:
    #
    #   prediction ± 1.96 sigma
    #
    # This is a diagnostic, not a claim of calibrated Gaussian
    # uncertainty.
    # ========================================================

    lower_95 = (
        mean_prediction
        -
        1.96 * predicted_sigma
    )


    upper_95 = (
        mean_prediction
        +
        1.96 * predicted_sigma
    )


    coverage_95 = np.mean(
        (
            y_test.to_numpy()
            >= lower_95
        )
        &
        (
            y_test.to_numpy()
            <= upper_95
        )
    )


    interval_width = np.mean(
        upper_95
        -
        lower_95
    )


    # ========================================================
    # PRINT PREDICTION RESULTS
    # ========================================================

    print()
    print("PREDICTION PERFORMANCE")

    print(
        "R²   =",
        round(
            r2,
            6
        )
    )

    print(
        "RMSE =",
        round(
            rmse,
            6
        )
    )

    print(
        "MAE  =",
        round(
            mae,
            6
        )
    )


    print()
    print("UNCERTAINTY")

    print(
        "95% coverage =",
        round(
            coverage_95 * 100,
            3
        ),
        "%"
    )

    print(
        "Mean interval width =",
        round(
            interval_width,
            6
        )
    )


    # ========================================================
    # STORE FOLD
    # ========================================================

    fold_results.append({

        "Fold":
            fold,

        "R2":
            r2,

        "RMSE":
            rmse,

        "MAE":
            mae,

        "Coverage_95":
            coverage_95,

        "Mean_Interval_Width":
            interval_width,

        "Mean_Predicted_Sigma":
            predicted_sigma.mean()
    })


    # ========================================================
    # STORE OBSERVATIONS
    # ========================================================

    meta = df.iloc[
        test_idx
    ][
        [
            "Code",
            "Lat",
            "Lon",
            "spatial_block"
        ]
    ].copy()


    meta["Fold"] = fold

    meta["Observed"] = (
        y_test.to_numpy()
    )


    meta["Predicted"] = (
        mean_prediction
    )


    meta["Absolute_Error"] = (
        absolute_error
    )


    meta["Squared_Error"] = (
        squared_error
    )


    meta["Predicted_Sigma"] = (
        predicted_sigma
    )


    meta["Predicted_Variance"] = (
        predicted_variance
    )


    meta["Lower_95"] = (
        lower_95
    )


    meta["Upper_95"] = (
        upper_95
    )


    meta["Covered_95"] = (
        (
            y_test.to_numpy()
            >= lower_95
        )
        &
        (
            y_test.to_numpy()
            <= upper_95
        )
    )


    observation_results.append(
        meta
    )


# ============================================================
# COMBINE
# ============================================================

fold_df = pd.DataFrame(
    fold_results
)

obs_df = pd.concat(
    observation_results,
    ignore_index=True
)

pearson_r = np.corrcoef(

q1 = uncertainty_summary.iloc[0]

    *
    100
)

print("============================================================")

print("============================================================")

print(
    "MAE degradation:",
    round(
        q4_q1_degradation,
        4
    ),
    "%"
)


# ============================================================
# RISK-COVERAGE ANALYSIS
# ============================================================
#
# Sort observations from LOW to HIGH predicted uncertainty.
#
# At each retained fraction, calculate the average actual error.
#
# If uncertainty is useful, removing the highest-uncertainty
# observations should preferentially remove high-error cases.
# ============================================================

sorted_obs = obs_df.sort_values(
    "Predicted_Sigma"
).reset_index(
    drop=True
)


coverage_levels = [
    1.00,
    0.90,
    0.80,
    0.70,
    0.60,
    0.50
]


risk_rows = []


for coverage_level in coverage_levels:

    n_keep = int(
        len(sorted_obs)
        * coverage_level
    )


    retained = sorted_obs.iloc[
        :n_keep
    ]


    risk_rows.append({

        "Coverage":
            coverage_level,

        "Observations_Retained":
            len(retained),

        "Mean_Predicted_Sigma":
            retained["Predicted_Sigma"].mean(),

        "Mean_Absolute_Error":
            retained["Absolute_Error"].mean(),

        "RMSE":
            np.sqrt(
                retained["Squared_Error"].mean()
            )
    })


risk_df = pd.DataFrame(
    risk_rows
)


print()
print("============================================================")
print("RISK-COVERAGE")
print("============================================================")

print(
    risk_df.to_string(
        index=False
    )
)


# ============================================================
# HIGH-RISK CONCENTRATION
# ============================================================
#
# What fraction of all absolute error lies in the highest 20%
# uncertainty observations?
# ============================================================

n_high = int(
    len(sorted_obs)
    * 0.20
)


high_uncertainty = sorted_obs.iloc[
    -n_high:
]


total_error = (
    obs_df["Absolute_Error"].sum()
)


high_error = (
    high_uncertainty[
        "Absolute_Error"
    ].sum()
)


high_error_share = (
    high_error
    /
    total_error
)


print()
print("============================================================")
print("HIGH-UNCERTAINTY ERROR CONCENTRATION")
print("============================================================")

print(
    "Highest 20% uncertainty observations:",
    len(high_uncertainty)
)

print(
    "Share of total absolute error:",
    round(
        high_error_share * 100,
        3
    ),
    "%"
)

        reg_lambda=1.0,

        objective="reg:squarederror",

        eval_metric="rmse",

        random_state=seed,

        n_jobs=-1,

        tree_method="hist"
    )


# ============================================================
# BUILD OUT-OF-FOLD RESIDUALS INSIDE OUTER TRAINING SET
# ============================================================
#
# This generates residual targets for uncertainty calibration.
#
# Every residual is produced by a model that did NOT train on
# the corresponding observation.
#
# The inner folds are spatial GroupKFold folds.
# ============================================================

def generate_internal_oof_residuals(
    X_outer_train,
    y_outer_train,
    groups_outer_train
):

    inner_cv = GroupKFold(
        n_splits=N_CALIBRATION_FOLDS
    )

    oof_prediction = np.full(
        len(X_outer_train),
        np.nan,
        dtype=float
    )


    for inner_fold, (
        inner_train_idx,
        inner_valid_idx
    ) in enumerate(

        inner_cv.split(
            X_outer_train,
            y_outer_train,
            groups=groups_outer_train
        ),

        start=1
    ):

        model = make_model(
            seed=RANDOM_STATE + inner_fold
        )


        model.fit(
            X_outer_train.iloc[
                inner_train_idx
            ],

            y_outer_train.iloc[
                inner_train_idx
            ]
        )


        oof_prediction[
            inner_valid_idx
        ] = model.predict(

            X_outer_train.iloc[
                inner_valid_idx
            ]
        )


    if not np.isfinite(
        oof_prediction
    ).all():

        raise RuntimeError(
            "Internal OOF predictions contain invalid values."
        )


    residual = (
        y_outer_train.to_numpy()
        -
        oof_prediction
    )


    squared_residual = (
        residual ** 2
    )


    absolute_residual = np.abs(
        residual
    )


    return (
        oof_prediction,
        residual,
        squared_residual,
        absolute_residual
    )


# ============================================================
# FIT FINAL MEAN MODEL
# ============================================================

def fit_mean_model(
    X_train,
    y_train
):

    model = make_model(
        seed=RANDOM_STATE
    )

    model.fit(
        X_train,
        y_train
    )

    return model


# ============================================================
# FIT UNCERTAINTY MODEL
# ============================================================
#
# The uncertainty target comes from INTERNAL OOF residuals.
#
# Therefore the uncertainty model is not trained on in-sample
# residuals.
# ============================================================

def fit_uncertainty_model(
    X_outer_train,
    y_outer_train,
    groups_outer_train
):

    (
        oof_prediction,
        residual,
        squared_residual,
        absolute_residual
    ) = generate_internal_oof_residuals(

        X_outer_train,
        y_outer_train,
        groups_outer_train
    )


    # --------------------------------------------------------
    # Clip only extreme residual targets.
    # Threshold is calculated inside outer training only.
    # --------------------------------------------------------

    clip_value = np.quantile(
        squared_residual,
        0.99
    )


    variance_target = np.minimum(
        squared_residual,
        clip_value
    )


    # Log transform.
    log_variance_target = np.log1p(
        variance_target
    )


    variance_model = make_model(
        seed=RANDOM_STATE + 1000
    )


    variance_model.fit(
        X_outer_train,
        log_variance_target
    )


    return (
        variance_model,
        clip_value
    )


# ============================================================
# CALIBRATE SIGMA USING INNER OOF DATA
# ============================================================
#
# We want a calibration map from raw predicted sigma to
# empirical absolute error.
#
# The calibrator is fitted only on OOF predictions/residuals
# generated from the outer training set.
#
# We use quantile binning rather than an external library so
# the procedure is completely transparent.
# ============================================================

def fit_sigma_calibrator(
    X_outer_train,
    y_outer_train,
    groups_outer_train,
    variance_model
):

    (
        oof_prediction,
        residual,
        squared_residual,
        absolute_residual
    ) = generate_internal_oof_residuals(

        X_outer_train,
        y_outer_train,
        groups_outer_train
    )


    raw_log_variance = (
        variance_model.predict(
            X_outer_train
        )
    )


    raw_variance = np.expm1(
        raw_log_variance
    )


    raw_variance = np.maximum(
        raw_variance,
        1e-8
    )


    raw_sigma = np.sqrt(
        raw_variance
    )


    calibration_frame = pd.DataFrame({

        "Raw_Sigma":
            raw_sigma,

        "Absolute_Error":
            absolute_residual

    })


    calibration_frame[
        "Sigma_Bin"
    ] = pd.qcut(

        calibration_frame[
            "Raw_Sigma"
        ],

        q=20,

        labels=False,

        duplicates="drop"
    )


    bin_table = (

        calibration_frame

        .groupby(
            "Sigma_Bin",
            observed=True
        )

        .agg(

            Mean_Raw_Sigma=(
                "Raw_Sigma",
                "mean"
            ),

            Mean_Absolute_Error=(
                "Absolute_Error",
                "mean"
            ),

            Observations=(
                "Raw_Sigma",
                "count"
            )
        )

        .reset_index()
    )


    # Ensure monotonic calibration mapping.
    #
    # Isotonic regression is preferable here because we do not
    # assume a particular parametric relationship.

    from sklearn.isotonic import IsotonicRegression


    isotonic = IsotonicRegression(
        increasing=True,
        out_of_bounds="clip"
    )


    isotonic.fit(

        bin_table[
            "Mean_Raw_Sigma"
        ],

        bin_table[
            "Mean_Absolute_Error"
        ],

        sample_weight=bin_table[
            "Observations"
        ]
    )


    return (
        isotonic,
        bin_table
    )


# ============================================================
# CREATE CALIBRATED TEST SIGMA
# ============================================================

def calibrated_test_sigma(
    X_test,
    variance_model,
    isotonic
):

    raw_log_variance = (
        variance_model.predict(
            X_test
        )
    )


    raw_variance = np.expm1(
        raw_log_variance
    )


    raw_variance = np.maximum(
        raw_variance,
        1e-8
    )


    raw_sigma = np.sqrt(
        raw_variance
    )


    calibrated_abs_error = (
        isotonic.predict(
            raw_sigma
        )
    )


    # --------------------------------------------------------
    # Convert expected absolute error into Gaussian-like sigma:
    #
    # E|e| ≈ sigma * sqrt(2/pi)
    #
    # This is a scale conversion only.
    # --------------------------------------------------------

    calibrated_sigma = (
        calibrated_abs_error
        /
        np.sqrt(
            2.0 / np.pi
        )
    )


    calibrated_sigma = np.maximum(
        calibrated_sigma,
        1e-6
    )


    return (
        raw_sigma,
        calibrated_sigma
    )


# ============================================================
# INTERVAL METRICS
# ============================================================

def interval_metrics(
    observed,
    prediction,
    sigma,
    z
):

    lower = (
        prediction
        -
        z * sigma
    )

    upper = (
        prediction
        +
        z * sigma
    )


    covered = (
        (observed >= lower)
        &
        (observed <= upper)
    )


    coverage = covered.mean()

    width = np.mean(
        upper - lower
    )


    return (
        coverage,
        width
    )

# ============================================================
#
# Every bootstrap sample resamples observations jointly.
#
# Therefore OOD, raw uncertainty and calibrated uncertainty
# always remain paired with the same high-error labels.
# ============================================================

rng = np.random.default_rng(
    RANDOM_STATE
)

bootstrap_cal_auc_diff = []

bootstrap_raw_auprc_diff = []
bootstrap_cal_auprc_diff = []


n = len(df)


for _ in range(
    N_BOOTSTRAPS

    indices = rng.integers(
        0,
        n,
        size=n
    )

    y_b = y[indices]

    if len(
        np.unique(y_b)

        continue

    ood_b = ood_score[
        indices
    ]

    raw_b = raw_uncertainty[
        indices
    ]

    cal_b = calibrated_uncertainty[
        indices
    ]


    auc_ood_b = roc_auc_score(
        y_b,
        ood_b
    )

    auc_raw_b = roc_auc_score(
        y_b,
        raw_b
    )

    auc_cal_b = roc_auc_score(
        y_b,
        cal_b
    )


    ap_ood_b = average_precision_score(
        y_b,
        ood_b
    )

    ap_raw_b = average_precision_score(
        y_b,
        raw_b
    )

    ap_cal_b = average_precision_score(
        y_b,
        cal_b
    )


    bootstrap_raw_auc_diff.append(
        auc_raw_b - auc_ood_b
    )

    bootstrap_cal_auc_diff.append(
        auc_cal_b - auc_ood_b

    bootstrap_raw_auprc_diff.append(

    values

    lower = np.percentile(
        values,
        2.5

    upper = np.percentile(
        values,
        97.5

    return lower, upper

# ============================================================
# SELECTIVE RISK FUNCTION
# ============================================================
#
# Higher score = more risk.
#
# For a requested coverage c:
#
#   keep the lowest-risk c fraction
#
#   reject the highest-risk (1-c) fraction
#
# ============================================================

def evaluate_selective_risk(
    score,
    coverage
):

    n = len(score)


    n_keep = max(
        1,
        int(
            np.floor(
                n * coverage
            )
        )
    )


    order = np.argsort(
        score
    )


    retained_indices = order[
        :n_keep
    ]


    rejected_indices = order[
        n_keep:
    ]


    retained = df.iloc[
        retained_indices
    ]


    rejected = df.iloc[
        rejected_indices
    ]


    retained_mae = retained[
        "Absolute_Error"
    ].mean()


    retained_rmse = np.sqrt(
        np.mean(
            (
                retained[
                    "Observed"
                ].to_numpy()
                -
                retained[
                    "Predicted"
                ].to_numpy()
            ) ** 2
        )
    )


    rejected_mae = np.nan

    rejected_rmse = np.nan


    if len(rejected) > 0:

        rejected_mae = rejected[
            "Absolute_Error"
        ].mean()


        rejected_rmse = np.sqrt(
            np.mean(
                (
                    rejected[
                        "Observed"
                    ].to_numpy()
                    -
                    rejected[
                        "Predicted"
                    ].to_numpy()
                ) ** 2
            )
        )


    error_reduction = (
        (
            baseline_mae
            -
            retained_mae
        )
        /
        baseline_mae
        *
        100
    )


    return {
        "Coverage":
            coverage,

# ============================================================
# OOD DIRECTION
# ============================================================

def get_ood_risk(
    y_event,
    ood_values
):

    positive_auc = roc_auc_score(
        y_event,
        ood_values
    )

    negative_auc = roc_auc_score(
        y_event,
        -ood_values
    )

    if positive_auc >= negative_auc:

        return (
            ood_values,
            "higher_score_more_risk"
        )

    return (
        -ood_values,
        "lower_score_more_risk"
    )


# ============================================================
# AURC CALCULATION
# ============================================================
#
# For every possible retained observation:
#
#   rank by risk
#   retain the lowest-risk observations
#   calculate selective MAE
#
# We then integrate risk over coverage using the trapezoidal
# rule.
#
# At full coverage the risk equals ordinary MAE.
# ============================================================

def calculate_aurc(
    errors,
    risk
):

    n = len(errors)

    order = np.argsort(
        risk
    )

    sorted_errors = errors[
        order
    ]


    cumulative_error = np.cumsum(
        sorted_errors
    )


    cumulative_n = np.arange(
        1,
        n + 1
    )


    coverage = (
        cumulative_n
        /
        n
    )


    selective_risk = (
        cumulative_error
        /
        cumulative_n
    )


    # Add the zero-coverage origin.
    coverage_curve = np.concatenate(
        [
            np.array([0.0]),
            coverage
        ]
    )


    risk_curve = np.concatenate(
        [
            np.array([
                selective_risk[0]
            ]),
            selective_risk
        ]
    )


    aurc = np.trapezoid(
        risk_curve,
        coverage_curve
    )


    return (
        aurc,
        coverage_curve,
        risk_curve
    )

def get_metric(
    dataframe,
    method,
    column
):

    rows = dataframe[
        dataframe["Method"] == method
    ]

    if rows.empty:
        return np.nan

    return float(
        rows[column].iloc[0]
    )


def require_column(
    dataframe,

):

    if column not in dataframe.columns:

            f"not found in {filename}.\n"
            f"Available columns: "
            f"{dataframe.columns.tolist()}"
        )

============================================================

import pandas as pd
import numpy as np

from sklearn.model_selection import GroupKFold
from sklearn.ensemble import IsolationForest
from sklearn.metrics import (
    r2_score,
    mean_squared_error,
    mean_absolute_error
)
from sklearn.preprocessing import StandardScaler

from xgboost import XGBRegressor
from scipy.stats import pearsonr, spearmanr


# ============================================================
# CONFIGURATION
# ============================================================

DATA_FILE = "data/YCEO_UHI_Enhanced_ML_Clean_Daytime.csv"

RANDOM_STATE = 42
N_FOLDS = 5

FEATURES = [
    "NDVI",
    "Elevation",
    "Builtup",
    "Population",
    "Mean_Temp_C",
    "Mean_Monthly_Precip_mm",
    "Urban_LC_Fraction",
    "Water_LC_Fraction",
    "Vegetation_LC_Fraction"
]

TARGET = "Annual_day"


# ============================================================
# LOAD DATA
# ============================================================

df = pd.read_csv(DATA_FILE)

df = df.dropna(
    subset=FEATURES + [
        TARGET,
        "Lat",
        "Lon"
    ]
).copy()

df = df[
    df["Population"] >= 0
].copy()

df = df[
    (df["NDVI"] >= -1) &
    (df["NDVI"] <= 1)
].copy()


# ============================================================
# SPATIAL BLOCKS
# ============================================================

df["lat_block"] = np.floor(
    df["Lat"] / 10
).astype(int)

df["lon_block"] = np.floor(
    df["Lon"] / 10
).astype(int)

df["spatial_block"] = (
    df["lat_block"].astype(str)
    + "_"
    + df["lon_block"].astype(str)
)


X = df[FEATURES]
y = df[TARGET]
groups = df["spatial_block"]


# ============================================================
# MODEL
# ============================================================

def make_model():

    return XGBRegressor(

        n_estimators=500,

        learning_rate=0.03,

        max_depth=6,

        min_child_weight=5,

        subsample=0.8,

        colsample_bytree=0.9,

        reg_lambda=1.0,

        objective="reg:squarederror",

        eval_metric="rmse",

        random_state=RANDOM_STATE,

        n_jobs=-1,

        tree_method="hist"
    )


# ============================================================
# 5-FOLD SPATIAL CV
# ============================================================

cv = GroupKFold(
    n_splits=N_FOLDS
)

fold_results = []
observation_results = []


for fold, (train_idx, test_idx) in enumerate(
    cv.split(
        X,
        y,
        groups=groups
    ),
    start=1
):

    print()
    print("================================================")
    print("SPATIAL FOLD", fold)
    print("================================================")


    X_train = X.iloc[
        train_idx
    ].copy()

    X_test = X.iloc[
        test_idx
    ].copy()

    y_train = y.iloc[
        train_idx
    ].copy()

    y_test = y.iloc[
        test_idx
    ].copy()


    # ========================================================
    # XGBOOST
    # ========================================================

    model = make_model()

    model.fit(
        X_train,
        y_train
    )

    prediction = model.predict(
        X_test
    )


    r2 = r2_score(
        y_test,
        prediction
    )

    rmse = np.sqrt(
        mean_squared_error(
            y_test,
            prediction
        )
    )

    mae = mean_absolute_error(
        y_test,
        prediction
    )


    print(
        "R²   =",
        round(r2, 6)
    )

    print(
        "RMSE =",
        round(rmse, 6)
    )

    print(
        "MAE  =",
        round(mae, 6)
    )


    # ========================================================
    # STANDARDIZE USING TRAINING DATA ONLY
    # ========================================================

    scaler = StandardScaler()

    train_scaled = scaler.fit_transform(
        X_train
    )

    test_scaled = scaler.transform(
        X_test
    )


    # ========================================================
    # ISOLATION FOREST
    # ========================================================
    #
    # Fit ONLY on training observations.
    #
    # Higher anomaly score = more unusual relative to
    # the training predictor distribution.
    # ========================================================

    detector = IsolationForest(

        n_estimators=300,

        contamination="auto",

        random_state=RANDOM_STATE,

        n_jobs=-1
    )

    detector.fit(
        train_scaled
    )


    # sklearn's decision_function:
    # higher = more normal.
    #
    # Therefore negate it so that:
    # higher OOD score = more unusual.
    #

    test_ood_score = -detector.decision_function(
        test_scaled
    )


    # ========================================================
    # ABSOLUTE ERROR
    # ========================================================

    absolute_error = np.abs(
        y_test.to_numpy()
        - prediction
    )


    # ========================================================
    # OBSERVATION RESULTS
    # ========================================================

    meta = df.iloc[
        test_idx
    ][
        [
            "Code",
            "Lat",
            "Lon",
            "spatial_block"
        ]
    ].copy()

    meta["Fold"] = fold

    meta["Observed"] = (
        y_test.to_numpy()
    )

    meta["Predicted"] = (
        prediction
    )

    meta["Absolute_Error"] = (
        absolute_error
    )

    meta["OOD_Score"] = (
        test_ood_score
    )

    observation_results.append(
        meta
    )


    fold_results.append({

        "Fold": fold,

        "R2": r2,

        "RMSE": rmse,

        "MAE": mae
    })


# ============================================================
# COMBINE
# ============================================================

fold_df = pd.DataFrame(
    fold_results
)

obs_df = pd.concat(
    observation_results,
    ignore_index=True
)


# ============================================================
# CORRELATION
# ============================================================

pearson_r, pearson_p = pearsonr(
    obs_df["OOD_Score"],
    obs_df["Absolute_Error"]
)

spearman_r, spearman_p = spearmanr(
    obs_df["OOD_Score"],
    obs_df["Absolute_Error"]
)


# ============================================================
# GLOBAL OOD QUARTILES
# ============================================================

obs_df["OOD_Quartile"] = pd.qcut(
    obs_df["OOD_Score"],
    q=4,
    labels=[
        "Q1 - most typical",
        "Q2",
        "Q3",
        "Q4 - most unusual"
    ],
    duplicates="drop"
)


ood_summary = (
    obs_df
    .groupby(
        "OOD_Quartile",
        observed=True
    )
    .agg(

        Observations=(
            "Code",
            "count"
        ),

        Mean_OOD_Score=(
            "OOD_Score",
            "mean"
        ),

        Mean_Absolute_Error=(
            "Absolute_Error",
            "mean"
        ),

        RMSE=(
            "Absolute_Error",
            lambda x:
            np.sqrt(
                np.mean(
                    x ** 2
                )
            )
        )
    )
    .reset_index()
)


# ============================================================
# WITHIN-FOLD OOD CORRELATIONS
# ============================================================

within_fold = []


for fold in sorted(
    obs_df["Fold"].unique()
):

    temp = obs_df[
        obs_df["Fold"] == fold
    ]

    p_r, p_p = pearsonr(
        temp["OOD_Score"],
        temp["Absolute_Error"]
    )

    s_r, s_p = spearmanr(
        temp["OOD_Score"],
        temp["Absolute_Error"]
    )


    within_fold.append({

        "Fold": fold,

        "Observations": len(temp),

        "Pearson_r": p_r,

        "Pearson_p": p_p,

        "Spearman_r": s_r,

        "Spearman_p": s_p
    })


within_fold_df = pd.DataFrame(
    within_fold
)


# ============================================================
# Q4-Q1 DEGRADATION
# ============================================================

q1 = ood_summary.iloc[0]
q4 = ood_summary.iloc[-1]


degradation = (
    (
        q4["Mean_Absolute_Error"]
        -
        q1["Mean_Absolute_Error"]
    )
    /
    q1["Mean_Absolute_Error"]
    * 100
)

print()
