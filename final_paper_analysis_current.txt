# ============================================================

def get_metric(
    dataframe,
    method,
    column
):

    rows = dataframe[
        dataframe["Method"] == method
    ]

    if rows.empty:
        return np.nan

    return float(
        rows[column].iloc[0]
    )


def require_column(
    dataframe,
    column,
    filename
):

    if column not in dataframe.columns:

        raise ValueError(
            f"Required column '{column}' "
            f"not found in {filename}.\n"
            f"Available columns: "
            f"{dataframe.columns.tolist()}"
        )
